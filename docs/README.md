```powershell
# CV 파이프라인으로 새 이미지 채점 (학습 불필요)
cd model_002
..\terring\Scripts\python.exe debug_chk.py 001.jpg   # 판정 시각화 + 점수 출력

# 85장 전수 평가 (라벨 대조 + 파라미터 그리드서치)
..\terring\Scripts\python.exe eval_cv.py

# CNN 학습 (5-fold 앙상블, 자동 좌표 라벨 필요)
cd ..\model_003
..\terring\Scripts\python.exe gen_labels.py    # CV로 좌표 라벨 생성
..\terring\Scripts\python.exe train_cnn.py     # 학습 + test 평가
```
